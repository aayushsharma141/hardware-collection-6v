```tsx
"use client";

import { useState } from "react";
import Image from "next/image";
import { motion, useReducedMotion } from "motion/react";
import { Brand, getSlugString } from "@/types/catalog";
import { useConsultationStore } from "@/components/consultation/store";

export interface BrandDiscoveryProps {
  brands: Brand[];
}

// UI-SPEC §6: "do not force 20+ brands through this heavy per-item hover choreography... the
// source brief's six-card scale this pattern was built for." Defensive cap, independent of how many
// documents an editor ever marks brand.featured in Studio.
const FEATURED_CAP = 6;

export default function BrandDiscovery({ brands }: BrandDiscoveryProps) {
  const shouldReduceMotion = useReducedMotion();
  const [hoveredId, setHoveredId] = useState<string | null>(null);
  const { openDrawer } = useConsultationStore();

  const featuredBrands = brands.filter((b) => b.featured).slice(0, FEATURED_CAP);
  const alphabeticalBrands = [...brands].sort((a, b) => a.name.localeCompare(b.name));

  const askAboutBrand = (brand: Brand) => {
    const slug = getSlugString(brand.slug) || brand.id || "";
    openDrawer({ source: "collections", intent: "consultation", brand: { slug, name: brand.name } });
  };

  return (
    <section aria-labelledby="brand-discovery-heading" className="py-16 md:py-20">
      {/* heading per Copywriting/Typography contract, then the two tiers below */}
      {/* ...Task 1 builds the featured tier, Task 2 builds the alphabetical wall... */}
    </section>
  );
}
```
